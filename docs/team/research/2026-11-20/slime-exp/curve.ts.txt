// Catmull-Rom-ish 근사 없이: 코사인 언덕 폴리라인(N 선분)
function hill(x0: number, x1: number, yBase: number, amp: number, n: number): Seg[] {
  const pts: Pt[] = []
  for (let i = 0; i <= n; i++) { const t = i / n; pts.push({ x: x0 + (x1 - x0) * t, y: yBase - amp * 0.5 * (1 - Math.cos(2 * Math.PI * t)) }) }
  return pts.slice(1).map((b, i) => ({ a: pts[i], b }))
}
